@echo off
set "CLANG_PATH=C:/clang+llvm-22.1.3-x86_64-pc-windows-msvc"
set "SOURCE_DIR=%CD%"
set "BUILD_DIR=%SOURCE_DIR%\build_custom"
set "INSTALL_DIR=%SOURCE_DIR%\install"

mkdir %BUILD_DIR%
cd %BUILD_DIR%

echo [Step 1] CMake Configuration (RelWithDebInfo + Static)
cmake -G "Ninja" ^
-DCMAKE_BUILD_TYPE=RelWithDebInfo ^
-DCMAKE_CXX_COMPILER="%CLANG_PATH%/bin/clang++.exe" ^
-DCMAKE_C_COMPILER="%CLANG_PATH%/bin/clang.exe" ^
-DBUILD_SHARED_LIBS=OFF ^
-DBUILD_WITH_STATIC_CRT=ON ^
-DCMAKE_INSTALL_PREFIX="%INSTALL_DIR%" ^
-DBUILD_EXAMPLES=OFF ^
-DBUILD_TESTS=OFF ^
-DBUILD_PERF_TESTS=OFF ^
-DWITH_IPP=OFF ^
-DBUILD_opencv_apps=OFF ^
..

echo [Step 2] Building...
ninja

echo [Step 3] Installing to %INSTALL_DIR%
ninja install

pause
